# Optional - only needed for the RAG pipeline (retriever.py, embeddings.py,
# build_faiss_index.py). Not required to run the app on the mock provider.
#
# torch is pinned to the CPU-only build explicitly (deployment-only concern -
# AthleteOS never uses a GPU). Without this pin, plain `pip install torch`
# (pulled in transitively by sentence-transformers) installs the default
# CUDA-enabled build, which drags in ~15 nvidia-cuda-* packages that are
# never used and meaningfully bloat both install size and process memory -
# a real problem on a memory-constrained host like Render's free tier.
# --extra-index-url below only ADDS a source pip searches; it never
# replaces PyPI, so every other package here still installs from PyPI as
# normal. Local dev is unaffected either way - installing torch==2.13.0+cpu
# works identically on a laptop.
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.13.0+cpu
sentence-transformers==5.6.1
faiss-cpu==1.15.0

# Used directly by retriever.py / build_faiss_index.py / embeddings.py for
# vector math (also an implicit dependency of the two packages above).
